<script src="https://unpkg.com/aos@2.3.4/dist/aos.js"></script>
<script>
AOS.init({ duration: 900, once: true });

const API = "https://api.mcsrvstat.us/3/play.breadsmp.my.id:55555";

const box = document.getElementById("server-box");
const statusText = document.getElementById("mc-status-text");
const ipText = document.getElementById("server-ip");
const countEl = document.getElementById("player-count");
const listEl = document.getElementById("player-list");

function loadServer(){
  fetch(API)
    .then(r=>r.json())
    .then(d=>{
      box.style.display="block";

      if(!d.online){
        statusText.innerHTML="🔴 Server Offline";
        countEl.innerHTML="";
        listEl.innerHTML="";
        return;
      }

      statusText.innerHTML="🟢 Server Online";
      ipText.innerHTML="🌐 IP: <b>play.breadsmp.my.id</b> • PORT: <b>55555</b>";
      countEl.innerHTML=`👥 Player: <b>${d.players.online}</b> / <b>${d.players.max}</b>`;
      listEl.innerHTML="";

      if(d.players.list){
        const javaPlayers=[];
        const bedrockPlayers=[];

        d.players.list.forEach(p=>{
          const name = typeof p==="string"?p:p.name;
          if(!name) return;

          if(name.startsWith(".")){
            bedrockPlayers.push(name.replace(".",""));
          }else{
            javaPlayers.push(name);
          }
        });

        [...javaPlayers,...bedrockPlayers].forEach(name=>{
          const isBedrock = bedrockPlayers.includes(name);
          const el=document.createElement("div");
          el.className="player-badge";
          el.innerHTML=`
            <img src="https://mc-heads.net/avatar/${name}/32">
            ${name}
            <span style="
              font-size:.7rem;
              padding:2px 6px;
              border-radius:6px;
              background:${isBedrock?'#22c55e':'#3b82f6'};
            ">
              ${isBedrock?'BEDROCK':'JAVA'}
            </span>
          `;
          listEl.appendChild(el);
        });
      }else{
        listEl.innerHTML="<i>Daftar player disembunyikan</i>";
      }
    })
    .catch(()=>{
      statusText.innerHTML="⚠️ Status tidak tersedia";
    });
}

loadServer();
setInterval(loadServer,5000);
</script>
